import json, os
from pathlib import Path
for proc in Path('/proc').iterdir():
    if proc.name.isdigit():
        try:
            if b'main.py' not in (proc/'cmdline').read_bytes(): continue
            for entry in (proc/'environ').read_bytes().split(b'\0'):
                if entry.startswith(b'DBUS_SESSION_BUS_ADDRESS='):
                    os.environ['DBUS_SESSION_BUS_ADDRESS']=entry.split(b'=',1)[1].decode()
        except OSError: pass
import gi
gi.require_version('Gtk','3.0')
from gi.repository import Gtk, Gdk, GLib
root=Path('/home/omnideck')
win=Gtk.Window(title='Visual desktop interaction test')
win.set_default_size(780,500); win.move(200,70)
canvas=Gtk.DrawingArea(); win.add(canvas)
targets={'source':[55,125,190,105], 'hover':[485,75,230,85], 'destination':[450,310,260,125]}
colors={'source':(.16,.40,.78), 'hover':(.50,.21,.68), 'destination':(.10,.48,.27)}
labels={'source':'DRAG SOURCE', 'hover':'HOVER HERE', 'destination':'DROP TARGET'}
state={'presses':[], 'releases':[], 'motions':[], 'hover':False}
def inside(key,x,y):
    bx,by,bw,bh=targets[key]; return bx<=x<bx+bw and by<=y<by+bh
def persist(): (root/'visual-live-events.json').write_text(json.dumps(state))
def draw(widget,cr):
    cr.set_source_rgb(.12,.14,.18); cr.paint()
    cr.set_source_rgb(1,1,1); cr.select_font_face('Sans'); cr.set_font_size(25)
    cr.move_to(28,45); cr.show_text('Native Linux: click, hover, drag')
    for key,rect in targets.items():
        cr.set_source_rgb(*colors[key]); cr.rectangle(*rect); cr.fill()
        cr.set_source_rgb(1,1,1); cr.set_font_size(22)
        cr.move_to(rect[0]+15,rect[1]+rect[3]/2+8); cr.show_text(labels[key])
    cr.set_font_size(18);cr.move_to(30,475)
    cr.show_text('Hover: %s | Presses: %s | Releases: %s'%(state['hover'],len(state['presses']),len(state['releases'])))
    return False
def event(widget,e):
    point=[round(e.x),round(e.y)]
    if e.type==Gdk.EventType.BUTTON_PRESS: state['presses'].append(point)
    elif e.type==Gdk.EventType.BUTTON_RELEASE: state['releases'].append(point)
    elif e.type==Gdk.EventType.MOTION_NOTIFY:
        state['hover']=inside('hover',e.x,e.y)
        if e.state&Gdk.ModifierType.BUTTON1_MASK: state['motions'].append(point)
    persist();canvas.queue_draw();return False
canvas.add_events(Gdk.EventMask.BUTTON_PRESS_MASK|Gdk.EventMask.BUTTON_RELEASE_MASK|Gdk.EventMask.POINTER_MOTION_MASK)
canvas.connect('draw',draw)
for name in ['button-press-event','button-release-event','motion-notify-event']:canvas.connect(name,event)
win.connect('destroy',Gtk.main_quit);win.show_all();win.present();persist()
def truth():
    x,y=canvas.get_window().get_origin()[-2:]
    (root/'visual-live-truth.json').write_text(json.dumps({'origin':[x,y],'targets':targets}));return False
GLib.timeout_add(700,truth)
Gtk.main()
