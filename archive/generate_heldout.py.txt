def button(ref,label,style='',icon=None):
 inner=f'<svg viewBox="0 0 24 24">{ICONS[icon]}</svg>' if icon else html.escape(label)
 return f'<button data-ref="{ref}" aria-label="{html.escape(label)}" style="{style}">{inner}</button>'
def shell(body,title='Workspace'):
 return f'<!doctype html><html><head><meta charset="utf-8"><style>{STYLE}</style></head><body><header>Desktop Lab · {title}</header><aside class="sidebar">Files<br>Recent<br>Projects<br>Downloads<br>Preferences</aside>{body}<div class="status">Ready · All changes saved</div></body></html>'
def parse_json(content):
 try: return json.loads(content.strip())
 except Exception:
  match=re.search(r'\{[^{}]*\}',content)
  if match:
   try: return json.loads(match.group())
   except Exception: pass
 return None

def call(fixture,mode,think=False,trial=1):
 path=SCENES/fixture['image']
 w,h=fixture['size']
 if mode=='overlay':
  path=SCENES/fixture['overlay']
  instruction='The screenshot has numbered badges marking clickable controls. Choose the badge for the requested target. Return only JSON {"ref":integer}. Do not return coordinates.'
 elif mode=='normalized':
  instruction=f'The screenshot is {w} pixels wide and {h} pixels high. Return the target center in normalized coordinates from 0 to 1000 on EACH axis: top-left is (0,0), bottom-right is (1000,1000). Return only JSON {{"x":integer,"y":integer}}. Use normalized coordinates, not pixels.'
 else:
  instruction=f'The screenshot is EXACTLY {w} pixels wide and {h} pixels high. Return the center of the target in ORIGINAL SCREENSHOT PIXEL coordinates: x ranges from 0 to {w-1}, y from 0 to {h-1}, origin at top-left. Return only JSON {{"x":integer,"y":integer}}. Do not use normalized coordinates.'
 prompt=fixture['task']+'\n'+instruction
 payload={'model':MODEL,'messages':[{'role':'user','content':prompt,'images':[base64.b64encode(path.read_bytes()).decode()]}],
          'stream':False,'think':think,'options':{'temperature':0,'num_predict':4096 if think else 512}}
 start=time.monotonic()
 try:
  res=requests.post(HOST+'/api/chat',json=payload,timeout=(10,110)); raw=res.json()
  result={'fixture':fixture['id'],'category':fixture['category'],'mode':mode,'think':think,'trial':trial,'status':res.status_code,'seconds':round(time.monotonic()-start,3),'prompt':prompt,'raw':raw}
 except Exception as exc:
  result={'fixture':fixture['id'],'category':fixture['category'],'mode':mode,'think':think,'trial':trial,'error':str(exc),'seconds':round(time.monotonic()-start,3)}
 content=result.get('raw',{}).get('message',{}).get('content',''); parsed=parse_json(content)
 result.update({'parsed':parsed,'strict_json':False,'hit':False,'point':None,'center_error_px':None})
 try: json.loads(content.strip()); result['strict_json']=True
 except Exception: pass
 if parsed:
  if mode=='overlay':
   ref=parsed.get('ref'); box=fixture['boxes'].get(str(ref))
   if box: point=[box['x']+box['width']/2,box['y']+box['height']/2]
   else: point=None
   result['correct_ref']=str(ref)==str(fixture['target_ref'])
  else:
   x,y=parsed.get('x'),parsed.get('y')
   point=[x,y] if isinstance(x,(int,float)) and isinstance(y,(int,float)) else None
   if point and mode=='normalized': point=[round(x*(w-1)/1000),round(y*(h-1)/1000)]
  if point:
   target=fixture['boxes'][str(fixture['target_ref'])]; x,y=point
   result['point']=point
   result['hit']=target['x']<=x<target['x']+target['width'] and target['y']<=y<target['y']+target['height']
   result['center_error_px']=round(math.hypot(x-(target['x']+target['width']/2),y-(target['y']+target['height']/2)),2)
 RESULTS.append(result)
 with (OUT/'results.jsonl').open('a') as stream: stream.write(json.dumps(result)+'\n')
 print(json.dumps({'fixture':fixture['id'],'category':fixture['category'],'mode':mode,'think':think,'hit':result['hit'],'point':result['point'],'error_px':result['center_error_px'],'seconds':result['seconds'],'content':content[:180]}),flush=True)
 return result
